

[밑줄 애니 css코드]

> 오버시 왼쪽에서 들어와서
  아웃시 오른쪽으로 나가는


/* 링크박스 a요소 */
.link-box a {
  /* 가상요소 자식이 block이므로 최소한
    display를 inline-block로 해야 자신의 크기만큼
    크기를 가지게 된다
    왜? inline내부의 block은 inline부모를 무시하고
    바깥박스를 기준하여 끝까지 width를 차지한다 */
  display: inline-block;
  font-size: 24px;
  line-height: 37px;
  color: #fff;
}

/* 가상 요소로 애니메이션 밑줄 만들기 */
.link-box a::after {
  content: "";
  display: block;
  /* 처음에 가로크기0 */
  width: 0%;
  /* 마진값으로 돌아올때 오른쪽으로 나가는 것 같은 효과 */
  /* width 마진을 %로 주면 부모박스 기준으로 ~ % 다 */
  margin-left: 100%;

  height: 1px;
  background-color: #5e70f5;
  /* 트랜지션 */
  transition: 0.3s linear;
}

/* a에 마우스 오버시 가상요소 밑줄 효과 */
.link-box a:hover::after {
  width: 100%;
  height: 1px;

  /* 왼쪽 마진값은 오버시 0 */
  margin-left: 0%;

  /* 트랜지션 */
  transition: width 0.3s linear;

  background-color: #5e70f57b;
}
